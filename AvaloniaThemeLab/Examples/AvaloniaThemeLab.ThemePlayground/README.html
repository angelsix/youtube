Overview

What you are building

The full interactive control gallery and live theme editor: extends DefaultTheme with a downstream BrandTheme, auto-discovers every themed control's {Name}Preview template, renders Light and Dark side by side with a live accent swatch picker, and opens a live theme settings window.

dotnet new install Avalonia.Templates

Phase 1 · Project & Packages

Create the project and install the packages

In addition to the default theme and engine, Theme Playground references AngelSix.ThemeEngine.Controls for LightDarkPreview and the themed calendar control.

  1. First we create a fresh Avalonia MVVM app and solution so we have a clean host for the full interactive theme playground.

    dotnet new avalonia.mvvm -n AvaloniaThemeLab.ThemePlayground
dotnet new sln -n AvaloniaThemeLab.ThemePlayground --format sln
dotnet sln AvaloniaThemeLab.ThemePlayground.sln add AvaloniaThemeLab.ThemePlayground/AvaloniaThemeLab.ThemePlayground.csproj

    Open AvaloniaThemeLab.ThemePlayground.sln in Rider rather than opening the folder — Rider reads the theme engine's live templates from the solution's .DotSettings file.

  2. Because we're using the AngelSix Theme Engine and AngelSix.Mvvm, we remove Avalonia's Fluent2Theme (or FluentTheme on older templates), the Inter font package, and CommunityToolkit.Mvvm so they don't conflict with our theme or MVVM base classes.

    dotnet remove AvaloniaThemeLab.ThemePlayground package Avalonia.Themes.Fluent2
dotnet remove AvaloniaThemeLab.ThemePlayground package Avalonia.Themes.Fluent
dotnet remove AvaloniaThemeLab.ThemePlayground package Avalonia.Fonts.Inter
dotnet remove AvaloniaThemeLab.ThemePlayground package CommunityToolkit.Mvvm

    Newer Avalonia 12 templates reference Avalonia.Themes.Fluent2 while older ones reference Avalonia.Themes.Fluent — remove whichever is in your .csproj. Removing CommunityToolkit.Mvvm ensures ViewModelBase and ObservableObject resolve to AngelSix.Mvvm. The theme engine styles controls directly and brings its own Inter font package (AngelSix.ThemeEngine.Fonts).

  3. Next we add DefaultTheme, ThemeEngine, ThemeEngine.Controls (which provides LightDarkPreview), and Microsoft's DI container.

    dotnet add AvaloniaThemeLab.ThemePlayground package AngelSix.Themes.Default
dotnet add AvaloniaThemeLab.ThemePlayground package AngelSix.ThemeEngine
dotnet add AvaloniaThemeLab.ThemePlayground package AngelSix.ThemeEngine.Controls
dotnet add AvaloniaThemeLab.ThemePlayground package Microsoft.Extensions.DependencyInjection

  4. Now we remove .WithInterFont() from Program.cs so the app compiles cleanly and lets the theme register its own Inter font.

Phase 3 · Preview Converters & Catalogue Discovery

Auto-discover every themed control and its preview template

Instead of maintaining a hard-coded list of controls, GalleryCatalogue inspects every loaded ControlTheme and pairs it with its {Name}Preview resource template.

  1. Our preview chrome needs two small value converters to map accent selections and theme state in XAML, so we add those helpers next.

    ValueConverters/PreviewTemplateConverter.csNew file
    using System;
using System.Globalization;
using Avalonia;
using Avalonia.Controls;
using Avalonia.Data.Converters;

namespace AvaloniaThemeLab.ThemePlayground.ValueConverters;

/// <summary>Finds a control type's gallery preview.</summary>
public sealed class PreviewTemplateConverter : IValueConverter
{
    public object? Convert(object? value, Type targetType, object? parameter, CultureInfo culture) =>
        value is Type controlType && Application.Current is { } app &&
        app.TryFindResource($"{controlType.Name}Preview", app.ActualThemeVariant, out var template)
            ? template
            : null;

    public object? ConvertBack(object? value, Type targetType, object? parameter, CultureInfo culture) =>
        throw new NotImplementedException();
}

    ViewModels/GalleryCatalogue.csNew file
    using AngelSix.ThemeEngine.Generated;
using System;
using System.Collections.Generic;
using System.Collections.ObjectModel;
using System.Diagnostics;
using System.Linq;
using System.Reflection;
using Avalonia;
using Avalonia.Controls;
using Avalonia.Controls.Templates;
using Avalonia.Styling;
using AngelSix.Themes.Default;

namespace AvaloniaThemeLab.ThemePlayground.ViewModels;

/// <summary>Discovers themed controls and their matching gallery pages from the loaded theme.</summary>
public sealed class GalleryCatalogue
{
    private readonly IReadOnlyList<GalleryEntryViewModel> _allEntries;

    /// <summary>The catalogue entries currently shown by the main window.</summary>
    public ObservableCollection<GalleryEntryViewModel> Entries { get; }

    public GalleryCatalogue(IEnumerable<PageEntryViewModel> pages)
    {
        var app = Application.Current ?? throw new InvalidOperationException("The gallery catalogue requires the Avalonia application to be initialized.");
        var specializedPages = pages.ToDictionary(page => page.ControlType);
        _allEntries = DiscoverEntries(app, specializedPages);
        Entries = new ObservableCollection<GalleryEntryViewModel>(_allEntries);
        RegisterMissingPreviewTemplate(app);
    }

    private GalleryCatalogue(IReadOnlyList<GalleryEntryViewModel> entries)
    {
        _allEntries = entries;
        Entries = new ObservableCollection<GalleryEntryViewModel>(entries);
    }

    public void Filter(string? query)
    {
        var matches = string.IsNullOrWhiteSpace(query)
            ? _allEntries
            : _allEntries.Where(entry => entry.Name.Contains(query, StringComparison.OrdinalIgnoreCase)).ToArray();

        Entries.Clear();
        foreach (var match in matches)
            Entries.Add(match);
    }

    private static IReadOnlyList<GalleryEntryViewModel> DiscoverEntries(
        Application app,
        IReadOnlyDictionary<Type, PageEntryViewModel> specializedPages)
    {
        var entries = new List<GalleryEntryViewModel>();
        var mistakes = new List<string>();

        foreach (var controlType in DiscoverThemedControlTypes(app).OrderBy(type => type.Name, StringComparer.Ordinal))
        {
            // The {Name}Preview template in the theme's control file is the single source of truth for what a control looks like, so the gallery prefers it.
            if (HasPreviewTemplate(app, controlType))
            {
                entries.Add(new ControlEntryViewModel(controlType));
            }
            else if (specializedPages.TryGetValue(controlType, out var page))
            {
                entries.Add(page);
            }
            else if (HasShownInMarker(app, controlType) || HasExcludedMarker(app, controlType))
            {
                // Deliberately excluded, and the theme file says so.
            }
            else
            {
                // A mistake: the control has a theme, so it is a control the gallery should show, but it has neither a preview nor a declaration of why it has none.
                var name = controlType.FullName ?? controlType.Name;
                mistakes.Add(name);
                entries.Add(new MissingPreviewEntryViewModel(controlType));
            }
        }

        var ordered = entries.OrderBy(entry => entry.Name, StringComparer.Ordinal).ToArray();
        ReportMistakes(mistakes, ordered);
        return ordered;
    }

    /// <summary>Says loudly, on stderr and in the debug output, which themed controls the gallery cannot show and why.</summary>
    private static void ReportMistakes(IReadOnlyList<string> mistakes, IReadOnlyList<GalleryEntryViewModel> entries)
    {
        if (mistakes.Count == 0)
            return;

        foreach (var mistake in mistakes)
        {
            Console.Error.WriteLine($"[gallery] MISSING PREVIEW: {mistake} has a ControlTheme but no " +
                                    "{Name}Preview template and no {Name}PreviewShownIn / {Name}PreviewExcluded marker. " +
                                    "It is listed in the gallery flagged as missing a preview.");
        }

        var message =
            $"[gallery] {mistakes.Count} themed control(s) have NO gallery preview and NO marker declaring why, " +
            "so they vanish from the gallery silently: " + string.Join(", ", mistakes) + ". " +
            $"Each one needs a marker in its own theme file: " +
            $"<theme:PreviewShownIn x:Key=\"<Name>PreviewShownIn\" ShownIn=\"{{x:Type <host>}}\" Reason=\"…\" /> " +
            "(shown inside another entry) or <theme:PreviewExcluded x:Key=\"<Name>PreviewExcluded\" Reason=\"…\" /> " +
            "(shown nowhere, and cannot have an entry of its own).";

        Console.Error.WriteLine(message);
        Debug.WriteLine(message);
    }

    private static IEnumerable<Type> DiscoverThemedControlTypes(Application app) =>
        AppDomain.CurrentDomain.GetAssemblies()
            .Where(assembly => !assembly.IsDynamic)
            .SelectMany(GetLoadableTypes)
            .Where(type => type.IsClass && !type.IsAbstract && !type.ContainsGenericParameters)
            .Where(type => typeof(Control).IsAssignableFrom(type))
            .Where(type => app.TryFindResource(type, app.ActualThemeVariant, out var resource) && resource is ControlTheme)
            .Distinct();

    private static Type[] GetLoadableTypes(Assembly assembly)
    {
        try
        {
            return assembly.GetTypes();
        }
        catch (ReflectionTypeLoadException exception)
        {
            return exception.Types.OfType<Type>().ToArray();
        }
    }

    private static bool HasPreviewTemplate(Application app, Type controlType) =>
        app.TryFindResource($"{controlType.Name}Preview", app.ActualThemeVariant, out var resource) && resource is IDataTemplate;

    /// <summary>Gives a flagged row something to open.</summary>
    private static void RegisterMissingPreviewTemplate(Application app)
    {
        if (app.DataTemplates.Any(template => template is MissingPreviewTemplate))
            return;
        app.DataTemplates.Add(new MissingPreviewTemplate());
    }

    /// <summary>Whether the control's theme file declares it as shown inside another entry.</summary>
    private static bool HasShownInMarker(Application app, Type controlType) =>
        app.TryFindResource($"{controlType.Name}PreviewShownIn", app.ActualThemeVariant, out var resource) &&
        resource is PreviewShownIn;

    /// <summary>Whether the control's theme file declares that it can have no gallery entry at all (a TopLevel, a popup host, an adorner layer).</summary>
    private static bool HasExcludedMarker(Application app, Type controlType) =>
        app.TryFindResource($"{controlType.Name}PreviewExcluded", app.ActualThemeVariant, out var resource) &&
        resource is PreviewExcluded;
}

/// <summary>A themed control the gallery cannot show, because the theme gives it no preview template and declares no reason for the omission.</summary>
public sealed class MissingPreviewEntryViewModel : GalleryEntryViewModel
{
    /// <summary>The words every flagged row carries, so a reviewer can search for them.</summary>
    public const string Flag = "missing preview";

    /// <summary>The control whose preview is missing.</summary>
    public Type ControlType { get; }

    public MissingPreviewEntryViewModel(Type controlType)
        : base($"{controlType.Name} — {Flag}")
    {
        ControlType = controlType;
    }

    /// <summary>What the page host shows when this row is selected.</summary>
    public override string ToString() =>
        $"MISSING PREVIEW: {ControlType.FullName ?? ControlType.Name} has a ControlTheme but no " +
        "{Name}Preview template, and its theme file declares no {Name}PreviewShownIn or " +
        "{Name}PreviewExcluded marker. It therefore has no gallery entry of its own. Add the marker " +
        "to its theme file to make the exclusion deliberate.";
}

/// <summary>The template a flagged gallery row opens: the page that says what is missing.</summary>
public sealed class MissingPreviewTemplate : IDataTemplate
{
    public Control Build(object? param) => new Pages.MissingPreviewPage();

    public bool Match(object? data) => data is MissingPreviewEntryViewModel;
}

  2. Now we add MainWindowViewModel to tie together the discovered control catalogue, search/selection state, and the popout theme editor.

    ViewModels/MainWindowViewModel.csNew file
    using System.Collections.Generic;
using System.Linq;
using AngelSix.Mvvm;
using AngelSix.ThemeEngine.Generated;

namespace AvaloniaThemeLab.ThemePlayground.ViewModels;

/// <summary>The gallery window: a searchable list of entries and the one being shown.</summary>
public sealed partial class MainWindowViewModel : ViewModelBase
{
    #region Fields

    // Absent in design-time and headless tools, which have no window to open over
    private readonly IThemeSettingsWindowService? _settingsWindow;

    // Armed while a navigating click on the sidebar is in flight (its press has landed, its release has not yet routed).
    private GalleryEntryViewModel? _revealedArmedFrom;

    #endregion Fields

    #region Properties

    /// <summary>The hues the header picker offers, in the order DefaultTheme declares them.</summary>
    public static IReadOnlyList<AccentHue> AccentChoices { get; } =
        [.. System.Enum.GetValues<AccentKind>().Where(kind => kind != AccentKind.Surface).Select(Hue)];

    public GalleryCatalogue Catalogue { get; }

    // Production default: the page is revealed only after the navigating click's release has routed (the whole point of the ShownEntry indirection).
    public bool DeferRevealPastNavigatingClick { get; set; } = true;

    #endregion Properties

    #region Observable Properties

    [Observable] public partial string SearchText { get; set; } = string.Empty;

    [Observable] public partial GalleryEntryViewModel? SelectedEntry { get; set; }

    [Observable] public partial GalleryEntryViewModel? ShownEntry { get; set; }

    // The app's one accent: the header picker's selection, styled into every preview.
    [Observable] public partial AccentHue SelectedAccent { get; set; } = Hue(AccentKind.Primary);

    #endregion Observable Properties

    #region Commands

    /// <summary>Opens the live theme editor over this window.</summary>
    public IRelayCommand OpenSettingsCommand => Command(() => _settingsWindow?.Show());

    #endregion Commands

    #region Constructors

    public MainWindowViewModel(GalleryCatalogue catalogue, IThemeSettingsWindowService? settingsWindow = null)
    {
        Catalogue = catalogue;
        _settingsWindow = settingsWindow;
        // No click is in flight, so the change handler shows the first entry straight away
        SelectedEntry = catalogue.Entries.FirstOrDefault();
    }

    #endregion Constructors

    #region Methods

    /// <summary>The engine hue for one of this theme's generated accent kinds.</summary>
    public static AccentHue Hue(AccentKind kind) => new(kind.ToString());

    partial void OnSelectedEntryChanged(GalleryEntryViewModel? value)
    {
        // While a navigating click is in flight its release still owns the pointer, so no other selection can happen meanwhile.
        if (DeferRevealPastNavigatingClick && _revealedArmedFrom is not null)
            return;
        ShownEntry = value;
    }

    /// <summary>Arms the hold-back for a navigating click whose press just landed.</summary>
    public void ArmDeferredReveal() => _revealedArmedFrom = ShownEntry;

    /// <summary>Disarms the hold-back and shows the selected entry now that the release has routed.</summary>
    public void CompleteDeferredReveal()
    {
        _revealedArmedFrom = null;
        ShownEntry = SelectedEntry;
    }

    // Keep the shown entry while it still matches; otherwise show the first match
    partial void OnSearchTextChanged(string value)
    {
        GalleryEntryViewModel? shown = SelectedEntry;
        Catalogue.Filter(value);

        SelectedEntry = shown is not null && Catalogue.Entries.Contains(shown)
            ? shown
            : Catalogue.Entries.FirstOrDefault();
    }

    #endregion Methods
}

  2. Next we register BrandTheme.Current and our ViewModels in DI inside App.axaml.cs and wire up the popout settings window.

    App.axaml.cs
    using System;
using AngelSix.Mvvm;
using AngelSix.ThemeEngine;
using AngelSix.Themes.Default;
using Avalonia;
using Avalonia.Controls;
using Avalonia.Controls.ApplicationLifetimes;
using Avalonia.Markup.Xaml;
using AvaloniaThemeLab.ThemePlayground.ViewModels;
using AvaloniaThemeLab.ThemePlayground.Views;
using Microsoft.Extensions.DependencyInjection;

namespace AvaloniaThemeLab.ThemePlayground;

public partial class App : Application, IServiceHost
{
    private ThemeContext? _themeContext;
    public IServiceProvider Services { get; private set; } = null!;

    public override void Initialize()
    {
        AvaloniaXamlLoader.Load(this);

        var active = ThemeContext.Resolver?.Invoke();
        _themeContext = ReferenceEquals(active?.Active, DefaultTheme.Current)
            ? active : new ThemeContext(DefaultTheme.Current);

        // After the load, so the theme App.axaml creates is there to register
        Services = BuildServices();
        DataTemplates.Add(Services.GetRequiredService<ViewLocator>());
    }

    public override void OnFrameworkInitializationCompleted()
    {
        if (ApplicationLifetime is IClassicDesktopStyleApplicationLifetime desktop)
        {
            desktop.MainWindow = new MainWindow { DataContext = Services.GetRequiredService<MainWindowViewModel>() };
            desktop.Exit += (_, _) =>
            {
                (_themeContext as IDisposable)?.Dispose();
                (Services as IDisposable)?.Dispose();
            };
        }

#if DEBUG
        // Dev-only: re-parse edited theme previews into the running app; no-op outside a dev checkout
        Development.PreviewHotReload.Start();
#endif

        base.OnFrameworkInitializationCompleted();
    }

    private static IServiceProvider BuildServices()
    {
        var services = new ServiceCollection();

        services.AddSingleton(DefaultTheme.Current);

        // The window a page's notifications are placed over
        services.AddSingleton<Func<TopLevel?>>(_ => () =>
            (Current?.ApplicationLifetime as IClassicDesktopStyleApplicationLifetime)?.MainWindow);

        services.AddSingleton<GalleryCatalogue>();
        services.AddSingleton<MainWindowViewModel>();
        services.AddSingleton<ThemeSettingsViewModel>();
        services.AddSingleton<IThemeSettingsWindowService>(sp => new ThemeSettingsWindowService(
            sp.GetRequiredService<Func<TopLevel?>>(), sp.GetRequiredService<ThemeSettingsViewModel>));
        // The Window page's dialogs and modal page
        services.AddSingleton<IDialogWindowService>(sp => new DialogWindowService(sp.GetRequiredService<Func<TopLevel?>>()));
        services.AddTransient<IModalNavigationService, ModalNavigationService>();
        services.AddTransient<WindowPageViewModel>();
        services.AddTransient<Pages.WindowPage>(sp => new Pages.WindowPage
        {
            DataContext = sp.GetRequiredService<WindowPageViewModel>(),
        });
        services.AddSingleton(sp => new ViewLocator(() => sp.GetRequiredService<Pages.WindowPage>()));
        services.AddSingleton<PageEntryViewModel>(sp => sp.GetRequiredService<WindowPageViewModel>());

        return services.BuildServiceProvider();
    }
}

  2. To keep clicking a sidebar item snappy without jumping mid-scroll, we add a small deferred reveal handler in MainWindow.axaml.cs.

    The sidebar ListBox flips SelectedEntry on pointer press; tunneling press/release ensures the new page reveals cleanly once the click finishes.

    Views/MainWindow.axaml.cs
    using System;
using Avalonia;
using Avalonia.Controls;
using Avalonia.Input;
using Avalonia.Interactivity;
using Avalonia.Threading;
using Avalonia.VisualTree;
using AvaloniaThemeLab.ThemePlayground.ViewModels;

namespace AvaloniaThemeLab.ThemePlayground.Views;

public partial class MainWindow : Window
{
    // True while a left-button press that began on the sidebar list is still held down.
    private bool _sidebarPressInFlight;

    public MainWindow()
    {
        InitializeComponent();

        // The sidebar selects on pointer press, so a clicked row flips SelectedEntry mid-click.
        AddHandler(InputElement.PointerPressedEvent, OnSidebarPressTunnel, RoutingStrategies.Tunnel);
        AddHandler(InputElement.PointerReleasedEvent, OnSidebarReleaseBubble, RoutingStrategies.Bubble);
    }

    /// <summary>Tunnels ahead of the sidebar item's own selection flip to mark a navigating press.</summary>
    private void OnSidebarPressTunnel(object? sender, PointerPressedEventArgs e)
    {
        // Only a left press counts as a navigation gesture (right/middle drive context menus etc.), and only one that begins somewhere inside the sidebar list.
        if (!e.Properties.IsLeftButtonPressed || e.Source is not Visual source)
            return;
        // Walk the press origin up its visual chain.
        foreach (var ancestor in source.GetSelfAndVisualAncestors())
        {
            if (ReferenceEquals(ancestor, SidebarEntries))
            {
                ArmSidebarNavigate();
                return;
            }
        }
    }

    /// <summary>Fires once the navigating press's release has routed; reveals the page past it.</summary>
    private void OnSidebarReleaseBubble(object? sender, PointerReleasedEventArgs e)
    {
        if (!_sidebarPressInFlight)
            return;
        FinishSidebarNavigate();
    }

    private void ArmSidebarNavigate()
    {
        _sidebarPressInFlight = true;
        if (DataContext is MainWindowViewModel vm)
            vm.ArmDeferredReveal();
    }

    private void FinishSidebarNavigate()
    {
        _sidebarPressInFlight = false;
        var vm = DataContext as MainWindowViewModel;
        if (vm is null)
            return;

        Dispatcher.UIThread.Post(() =>
        {
            if (_sidebarPressInFlight)
                return; // Superseded by a newer press; that one owns the reveal
            vm.CompleteDeferredReveal();
        }, DispatcherPriority.Normal);
    }

}
